## Global Constraints

- Commit messages follow Conventional Commits and carry **no** attribution lines: no `Co-Authored-By`, no "Generated with" footer, nothing naming Claude or Anthropic.
- Use `yarn` (3.6.1) for every script in this repo, never `npm`.
- Never bypass the hooks (`--no-verify`). Lefthook runs Prettier, ESLint and `tsc` on staged files and commitlint on the message; fix what they report.
- Stage only the files the task names (`git add <paths>`), never `git add -A`.
- Do not push, tag or publish. Do not touch `src/assets/fonts` or any brand asset.
- Keep the public API stable. The only additions are `blockTouchesWhileVisible` on `NotificationProvider`, `blockTouchesWhileNotificationVisible` on `RottProvider`, and an optional `onPress` on `NotificationModel`.
- The repo's Jest setup cannot render `react-native-tab-view` or a real toast: showing either loads React Native's legacy renderer, which rejects `react@19.2.4` because it is built for `19.1.0`. Tests for those two stub the library in their own file, as written below.
- After a code change, update the component's English docs page. Update `docs/i18n/tr` only where a mirror of that page exists (`theming/rott-config.md` in this plan).

- [ ] **Step 5: Replace `src/libs/i18n/utils/formatMessage.ts`**

```ts
import {
  createIntl,
  createIntlCache,
  type IntlShape,
  type MessageDescriptor,
  type PrimitiveType,
} from 'react-intl'

import '../../../utils/arrayExtensions'
import '../../../utils/stringExtensions'

import {languageMessages} from '../constants'
import {getActiveLanguage} from './activeLanguage'
import type {TranslationLanguageTypes} from './models'

const intlCache = createIntlCache()
const intlByLocale = new Map<string, IntlShape>()

const getIntl = (locale: string): IntlShape => {
  const cached = intlByLocale.get(locale)
  if (cached) return cached

  const messages = Object.prototype.hasOwnProperty.call(languageMessages, locale)
    ? languageMessages[locale as keyof typeof languageMessages]
    : languageMessages['en-US']
  const intl = createIntl({locale, messages}, intlCache)
  intlByLocale.set(locale, intl)

  return intl
}

/**
 * Translates a key from the library's language files into the active language.
 *
 * Not a hook: it can be called during render, inside event handlers, or outside React.
 * The active language follows the most recently rendered RottProvider and defaults to en-US.
 *
 * @param {TranslationLanguageTypes} message The key written in the language file. Example: Common.Hello -> Outputs "Merhaba" in Turkish, "Hello" in English.
 * @param {Record<string, PrimitiveType | any | undefined>} extraMessages Values for the placeholders in the message. Example: {name: 'Cindoruk'}.
 * @param {MessageDescriptor} descriptor A full descriptor to use instead of the key. Example: {id: 'Common.Hello', defaultMessage: 'Hello'}.
 *
 * @returns {string} The translation in the active language. A key without a translation is returned as is.
 */
export const formatMessage = (
  message: TranslationLanguageTypes,
  extraMessages?: Record<string, PrimitiveType | any | undefined>,
  descriptor?: MessageDescriptor
): string =>
  getIntl(getActiveLanguage()).formatMessage(descriptor ?? {id: `${message}`}, extraMessages)
```

```ts
/** The shape AmountInput emits through onChangeText: integer digits, a dot, one or two decimals. */
const EMITTED_AMOUNT = /^(\d+)\.(\d{1,2})$/

/** Groups an integer's digits in threes with dots, the way the amount field displays them. */
export const groupThousands = (digits: string): string =>
  digits.replace(/\B(?=(\d{3})+(?!\d))/g, '.')

/**
 * Splits an AmountInput value into the display text of its two fields.
 *
 * Accepts the component's own output ("1234.50") and the Turkish-formatted input it has
 * always read ("1.234,50"), so mounting with a value the component emitted is stable.
 */
export const parseAmountValue = (value?: string | null): {amount: string; currency: string} => {
  if (!value) return {amount: '', currency: ''}

  const emitted = EMITTED_AMOUNT.exec(value)
  if (emitted) return {amount: groupThousands(emitted[1] ?? ''), currency: emitted[2] ?? ''}

  const [amount = '', currency = ''] = value.split(',')

  return {amount, currency}
}
```

````md
## Value Format

`onChangeText` receives the amount as a dot-decimal string with two decimals, such as
`"1234.50"`. You can pass that string straight back as `value`. The input also accepts
the Turkish-formatted text it displays, such as `"1.234,50"`.

```tsx
const [amount, setAmount] = useState('1234.50')

<Input name='amount' type='amount' value={amount} onChangeText={setAmount} />
````

```tsx
import {
  NotificationContext,
  NotificationModel,
  notificationRef,
  ToasterNotificationProvider,
} from '..'
import {Item} from '../../Item'
import {useNotification} from '../hooks'

import {Toast} from 'react-native-toast-notifications'

const InitializeNotificationRef = () => {
  const notificationHook = useNotification()
  notificationRef.current = notificationHook

  return null
}

export interface NotificationProviderProps extends PropsWithChildren {
  /**
   * When true, the whole tree ignores touches while a notification is visible — the
   * behavior before 0.9.1. Off by default, so the app and the toast stay pressable.
   */
  blockTouchesWhileVisible?: boolean
}

export const NotificationProvider: FC<NotificationProviderProps> = ({
  children,
  blockTouchesWhileVisible = false,
}) => {
  // Toasts on screen. The toast library calls onClose once per toast — hidden by id, by
  // hideAll or by its timer — so onClose is the only place that counts down.
  const [visibleNotifications, setVisibleNotifications] = useState(0)

  const handleClose = useCallback(() => {
    setVisibleNotifications((count) => Math.max(0, count - 1))
  }, [])

  const show = useCallback(
    ({title, description, variant, onPress}: NotificationModel) => {
      setVisibleNotifications((count) => count + 1)

      return Toast.show(title, {
        type: variant,
        data: {title, description, variant, onPress},
        onClose: handleClose,
      })
    },
    [handleClose]
  )

  const hide = useCallback((id: string) => Toast.hide(id), [])
  const hideAll = useCallback(() => Toast.hideAll(), [])

  const success = useCallback(
    (title: string, description?: string, onPress?: () => void) =>
      show({title, description, variant: 'success', onPress}),
    [show]
  )
  const warning = useCallback(
    (title: string, description?: string, onPress?: () => void) =>
      show({title, description, variant: 'warning', onPress}),
    [show]
  )
  const danger = useCallback(
    (title: string, description?: string, onPress?: () => void) =>
      show({title, description, variant: 'danger', onPress}),
    [show]
  )
  const info = useCallback(
    (title: string, description?: string, onPress?: () => void) =>
      show({title, description, variant: 'info', onPress}),
    [show]
  )

  const contextValues = useMemo(
    () => ({
      notifications: visibleNotifications,
      show,
      hide,
      hideAll,
      success,
      warning,
      danger,
      info,
    }),
    [visibleNotifications, show, hide, hideAll, success, warning, danger, info]
  )

  const blockTouches = blockTouchesWhileVisible && visibleNotifications > 0

  return (
    <NotificationContext.Provider value={contextValues}>
      <InitializeNotificationRef />
      <Item flex={1} pointerEvents={blockTouches ? 'box-only' : 'box-none'}>
        <ToasterNotificationProvider>{children}</ToasterNotificationProvider>
      </Item>
    </NotificationContext.Provider>
```

````md
# Notification

Notification shows short toast messages at the top of the screen. `RottProvider` sets

## Features

- 🎨 Four variants: success, warning, danger and info
- ⏱️ Auto-dismiss after three seconds

  navigation.navigate('Inbox')

| Method | Parameters | Description |
|--------|------------|-------------|
| `success` | `(title: string, description?: string, onPress?: () => void)` | Show a success toast |
| `warning` | `(title: string, description?: string, onPress?: () => void)` | Show a warning toast |
| `danger` | `(title: string, description?: string, onPress?: () => void)` | Show an error toast |
| `info` | `(title: string, description?: string, onPress?: () => void)` | Show an info toast |

| `hideAll` | `()` | Hide every visible toast |
````
